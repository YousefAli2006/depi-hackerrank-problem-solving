def arrayManipulation(n,queries):
    diff=[0]*(n+2)
    for a,b,k in queries:
        diff[a]+=k
        diff[b+1]-=k
    running=0
    max_val=0
    for i in range(1,n+1):
        running+=diff[i]
        if running>max_val:
            max_val=running
    return max_val

def dynamicArray(n,queries):
    seq_list = [[] for _ in range(n)]
    last_answer = 0
    answers = []
    for query in queries:
        q_type = query[0]
        x = query[1]
        y = query[2]
        idx=(x^last_answer)%n
        if q_type==1:
            seq_list[idx].append(y)
        else:
            seq=seq_list[idx]
            val=seq[y%len(seq)]
            last_answer=val
            answers.append(val)
    return answers

class Queue:
    def __init__(self):
        self.stack_in=[]
        self.stack_out=[]
    def enqueue(self,x):
        self.stack_in.append(x)
    def dequeue(self):
        if not self.stack_out:
            while self.stack_in:
                self.stack_out.append(self.stack_in.pop())
        self.stack_out.pop()
    def peek(self):
        if not self.stack_out:
            while self.stack_in:
                self.stack_out.append(self.stack_in.pop())
        return self.stack_out[-1]

class MinHeap:
    def __init__(self,initial=None):
        self.heap=[]
        if initial:
            self.heap=list(initial)
            self.heapify()
    def push(self,item):
        self.heap.append(item)
        self.sift_up(len(self.heap)-1)
    def pop(self):
        if not self.heap:
            raise IndexError("pop from empty heap")
        self.my_swap(0,-1)
        smallest=self.heap.pop()
        if self.heap:
            self.sift_down(0)
        return smallest
    def peek(self):
        if not self.heap:
            raise IndexError("peek from empty heap")
        return self.heap[0]
    def __len__(self):
        return len(self.heap)
    def heapify(self):
        for i in range(len(self.heap)//2-1,-1,-1):
            self.sift_down(i)
    def sift_up(self,idx):
        parent=(idx-1)//2
        while idx>0 and self.heap[idx]<self.heap[parent]:
            self.my_swap(idx,parent)
            idx=parent
            parent=(idx-1)//2
    def sift_down(self,idx):
        n=len(self.heap)
        while True:
            left=2*idx+1
            right=2*idx+2
            smallest=idx
            if left<n and self.heap[left]<self.heap[smallest]:
                smallest=left
            if right<n and self.heap[right]<self.heap[smallest]:
                smallest=right
            if smallest==idx:
                break
            self.my_swap(idx,smallest)
            idx=smallest
    def my_swap(self,i,j):
        self.heap[i],self.heap[j]=self.heap[j],self.heap[i]

import heapq
def cookies(k,arr):
    heapq.heapify(arr)
    n_operations=0
    while len(arr)>=2 and arr[0]<k:
        x=heapq.heappop(arr)
        y=heapq.heappop(arr)
        heapq.heappush(arr,x+2*y)
        n_operations+=1
    return n_operations if arr[0]>=k else -1

def isBalanced(s):
    stack=[]
    pairs={')':'(',']':'[','}':'{'}
    for c in s:
        if c in pairs:
            if not stack or stack[-1]!=pairs[c]:
                return False
            stack.pop()
        else:
            stack.append(c)
    return not stack

import sys
sys.setrecursionlimit(2000)

class Node:
    def __init__(self,val):
        self.val=val
        self.left=None
        self.right=None

class BinaryTree:
    def __init__(self,L,R):
        self.L=L
        self.R=R
        self.root=self.constructor_helper(1,1)

    def constructor_helper(self,i,d):
        if i==-1:
            return None
        node=Node(i)
        node.depth=d
        node.left=self.constructor_helper(self.L[i],d+1)
        node.right=self.constructor_helper(self.R[i],d+1)
        return node

    def swap(self,k):
        def helper_swap(node):
            if not node:
                return
            if node.depth%k==0:
                node.left,node.right=node.right,node.left
            helper_swap(node.left)
            helper_swap(node.right)
        helper_swap(self.root)

    def inorder(self):
        res=[]
        def helper_inorder(node):
            if not node:
                return
            helper_inorder(node.left)
            res.append(node.val)
            helper_inorder(node.right)
        helper_inorder(self.root)
        return res

n=int(input())
L=[0]*(n+1)
R=[0]*(n+1)
for i in range(1,n+1):
    l,r=map(int,input().split())
    L[i]=l
    R[i]=r

tree=BinaryTree(L,R)
q=int(input())
for _ in range(q):
    k=int(input())
    tree.swap(k)
    print(*tree.inorder())
