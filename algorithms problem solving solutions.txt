def compareTriplets(a,b):
    a_sum=0
    b_sum=0
    for i in range(3):
        if a[i]>b[i]:
            a_sum+=1
        elif a[i]<b[i]:
            b_sum+=1
    return [a_sum,b_sum]

def plusMinus(arr):
    positive_counter=0
    negative_counter=0
    zero_counter=0
    for i in arr:
        if i>0:
            positive_counter+=1
        elif i<0:
            negative_counter+=1
        else:
            zero_counter+=1
    return [positive_counter/len(arr),negative_counter/len(arr),zero_counter/len(arr)]
n=int(input())
arr=list(map(int,input().split(" ")))
print(*plusMinus(arr),sep="\n")

def birthday(s,d,m):
    ways=[]
    for i in range(len(s)):
        if sum(s[i:i+m])==d:
            ways.append(s[i:i+m])
    return len(ways)

def divisibleSumPairs(n,k,ar):
    pairs=[]
    for i in range(len(ar)):
        for j in range(len(ar)):
            if i<j and (ar[i]+ar[j])%k==0:
                pairs.append((i,j))
    return len(pairs)

ar=list(map(int,input().split(" ")))

    nums_reverse=map(lambda x:int(str(x)[::-1]),list(range(i,j+1)))

    e=100
    cloud_index=0
    while True:
        cloud_index=(cloud_index+k)%len(c)
        if c[cloud_index]==0:
            e-=1
        else:
            e-=3
        if cloud_index==0:
            break
    return e

import sys
sys.set_int_max_str_digits(0)
def fibonacciModified(t1,t2,n):
    if n==1:
        return t1
    if n==2:
        return t2
    for _ in range(n-2):
        t1,t2=t2,t1+t2**2
    return t2

def getWays(n,c):
    dp=[0]*(n+1)
    dp[0]=1
    for coin in c:
        for i in range(coin,n+1):
            dp[i]+=dp[i-coin]
    return dp[n]
n,m=map(int,input().split())
c=list(map(int,input().split()))
print(getWays(n,c))

from math import gcd
def getTotalX(a,b):
    my_lcm=a[0]
    for x in a[1:]:
        my_lcm=my_lcm*x//gcd(my_lcm,x)
    my_gcd=b[0]
    for x in b[1:]:
        my_gcd=gcd(my_gcd,x)
    c=0
    for i in range(my_lcm,my_gcd+1,my_lcm):
        if my_gcd%i==0:
            c+=1
    return c

n,m=map(int,input().split())
a=list(map(int,input().split()))
b=list(map(int,input().split()))
print(getTotalX(a,b))
